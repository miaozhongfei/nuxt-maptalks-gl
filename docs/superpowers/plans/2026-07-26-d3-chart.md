### Task 2: Create shared `createD3Viz.ts`

**Files:**
- Create: `examples/demo-suite/app/components/example2/style/d3-chart/createD3Viz.ts`

**Interfaces:**
- Produces: `export function createD3Viz(container: HTMLElement): void`

**Description:** 共享纯函数封装。接收 UIMarker 内部的 `.d3-container` div，注入 CSS、建 quadtree、渲染 SVG、挂 brush。

- [ ] **Step 1: Write `createD3Viz.ts`**

```ts
import { brush, quadtree, range, scaleIdentity, select } from 'd3'

const W = 600
const H = 300

/**
 * 在给定容器中创建 D3 四叉树 + 画刷交互可视化（对应官网 5.30）。
 *
 * @description 生成 5000 随机点，建 d3.quadtree，渲染节点矩形与数据点圆，
 * 挂载 d3.brush 框选，高亮 scanned/selected 点。
 * @param {HTMLElement} container - UIMarker 内的 .d3-container div
 *
 * @example
 * const dom = uiMarker.getDOM()?.querySelector('.d3-container') as HTMLElement
 * if (dom) createD3Viz(dom)
 */
export function createD3Viz(container: HTMLElement): void {
  container.innerHTML = ''
  const style = document.createElement('style')
  style.textContent = [
    '.point{fill:#999;stroke:#fff}',
    '.point.scanned{fill:orange;fill-opacity:1;stroke:brown}',
    '.point.selected{fill:red;fill-opacity:1}',
    '.node{fill:none;stroke:#ccc;shape-rendering:crispEdges}',
    '.brush .selection{stroke:#fff;fill-opacity:.125;shape-rendering:crispEdges}',
  ].join('')
  container.appendChild(style)

  // data: [x, y, scanned, selected]
  const data = range(5000).map(() => <const>[Math.random() * W, Math.random() * H, false, false])

  const quadtreeRoot = quadtree<readonly [number, number, boolean, boolean]>(
    data as [number, number, boolean, boolean][],
    (d) => d[0],
    (d) => d[1],
  ).extent([[-1, -1], [W + 1, H + 1]])

  const svg = select(container).append('svg').attr('width', W).attr('height', H)

  // 节点矩形
  const nodes: { x: number; y: number; width: number; height: number }[] = []
  quadtreeRoot.visit((_node, x1, y1, x2, y2) => {
    nodes.push({ x: x1, y: y1, width: x2 - x1, height: y2 - y1 })
    return false
  })
  svg.selectAll<SVGRectElement, typeof nodes[0]>('.node')
    .data(nodes)
    .enter().append('rect')
    .attr('class', 'node')
    .attr('x', (d) => d.x)
    .attr('y', (d) => d.y)
    .attr('width', (d) => d.width)
    .attr('height', (d) => d.height)

  // 数据点圆（re-select 因为 visit 可能修改 data）
  const circle = svg.selectAll<SVGCircleElement, typeof data[0]>('.point')
    .data(data as [number, number, boolean, boolean][])
    .enter().append('circle')
    .attr('class', 'point')
    .attr('cx', (d) => d[0])
    .attr('cy', (d) => d[1])
    .attr('r', 4)

  // 画刷
  const brushGen = brush<SVGGElement>()
    .extent([[0, 0], [W, H]])
    .on('brush', ({ selection }) => {
      if (!selection) return
      const [[x0, y0], [x1, y1]] = selection as [[number, number], [number, number]]
      const pts = data as [number, number, boolean, boolean][]
      // reset
      for (const p of pts) { p[2] = false; p[3] = false }
      // search
      quadtreeRoot.visit((_node, qx1, qy1, qx2, qy2) => {
        const n = _node as { data?: readonly [number, number, boolean, boolean] }
        const p = n.data
        if (p) {
          p[2] = true
          p[3] = p[0] >= x0 && p[0] < x1 && p[1] >= y0 && p[1] < y1
        }
        return qx1 >= x1 || qy1 >= y1 || qx2 < x0 || qy2 < y0
      })
      circle.classed('scanned', (d) => d[2])
      circle.classed('selected', (d) => d[3])
    })

  const g = svg.append('g').attr('class', 'brush').call(brushGen)
  g.call(brushGen.move as (g: unknown, s: [[number, number], [number, number]]) => void, [[100, 100], [200, 200]])
}
```

### Task 3: Rewrite EscapeImpl.vue

**Files:**
- Modify: `examples/demo-suite/app/components/example2/style/d3-chart/EscapeImpl.vue`

**Interfaces:**
- Consumes: `createD3Viz(container: HTMLElement): void`

- [ ] **Step 1: Rewrite EscapeImpl**

```vue
<template>
  <div
    ref="el"
    class="relative rounded border border-default overflow-hidden"
    style="height: 480px"
  />
</template>

<script setup lang="ts">
import { createD3Viz } from './createD3Viz'

const el = ref<HTMLElement | null>(null)
const { map } = useMaptalks(el, { center: [121.5057, 31.2453], zoom: 13 })
useMaptalksTileLayer(map, { source: 'osm' })

let uiMarker: { remove: () => void; getDOM: () => HTMLElement } | null = null

watch(
  () => toValue(map),
  async (m) => {
    if (!m) return
    const mt = await import('maptalks-gl')
    const uim = new (mt as unknown as { ui: { UIMarker: new (c: [number, number], o: Record<string, unknown>) => { addTo: (m: unknown) => void; remove: () => void; getDOM: () => HTMLElement } } }).ui.UIMarker(
      [121.5057, 31.2453],
      {
        content: '<div class="d3-container" style="width:600px;height:300px;background:#fff"></div>',
        single: false,
        draggable: false,
      },
    )
    uim.addTo(m)
    uiMarker = uim
    setTimeout(() => {
      const container = uim.getDOM()?.querySelector('.d3-container') as HTMLElement | null
      if (container) createD3Viz(container)
    }, 100)
  },
)

onBeforeUnmount(() => {
  uiMarker?.remove()
})
</script>
```
